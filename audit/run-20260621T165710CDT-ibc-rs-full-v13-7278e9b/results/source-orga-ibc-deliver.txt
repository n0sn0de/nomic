   167	    /// Execute one [IbcMessage].
   168	    ///
   169	    /// Returns info about a transfer if the message triggered one.
   170	    pub fn deliver_message(&mut self, message: IbcMessage) -> crate::Result<Option<TransferInfo>> {
   171	        let mut maybe_client_update = None;
   172	
   173	        use IbcMessage::*;
   174	
   175	        match message {
   176	            Ics26(msg) => {
   177	                if let MsgEnvelope::Client(ClientMsg::UpdateClient(msg)) = &msg {
   178	                    maybe_client_update = Some(msg.clone());
   179	                }
   180	                dispatch(&mut self.ctx, &mut self.router, msg)
   181	                    .map_err(|e| Error::Ibc(e.to_string()))?
   182	            }
   183	            Ics20(msg) => {
   184	                let transfer_module = &mut self.router.transfer;
   185	                send_transfer(&mut self.ctx, transfer_module, msg)
   186	                    .map_err(|e| Error::Ibc(e.to_string()))?
   187	            }
   188	        };
   189	
   190	        if let Some(msg) = maybe_client_update {
   191	            match Header::try_from(msg.client_message) {
   192	                Ok(header) => {
   193	                    let mut client = self
   194	                        .ctx
   195	                        .clients
   196	                        .get_mut(msg.client_id.into())?
   197	                        .ok_or_else(|| Error::Ibc("Expected client".to_string()))?;
   198	                    client.last_header = Some(header.into())
   199	                }
   200	                Err(err) => log::debug!("Error decoding header: {}", err),
   201	            }
   202	        }
