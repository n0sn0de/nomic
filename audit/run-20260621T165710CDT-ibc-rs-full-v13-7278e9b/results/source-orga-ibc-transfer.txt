   187	impl TokenTransferExecutionContext for Transfer {
   188	    fn burn_coins_execute(
   189	        &mut self,
   190	        account: &Self::AccountId,
   191	        coin: &PrefixedCoin,
   192	        _memo: &Memo,
   193	    ) -> Result<(), TokenTransferError> {
   194	        let denom: Denom = coin.denom.clone().try_into()?;
   195	        let amount: Amount = coin.amount.try_into()?;
   196	
   197	        let mut denom_balances = self.accounts.entry(denom)?.or_default()?;
   198	
   199	        let mut account_balance = denom_balances.entry(*account)?.or_default()?;
   200	        *account_balance = (*account_balance - amount).result()?;
   201	
   202	        Ok(())
   203	    }

   205	    fn mint_coins_execute(
   206	        &mut self,
   207	        account: &Self::AccountId,
   208	        coin: &PrefixedCoin,
   209	    ) -> Result<(), TokenTransferError> {
   210	        let denom: Denom = coin.denom.clone().try_into()?;
   211	        let amount: Amount = coin.amount.try_into()?;
   212	
   213	        let mut denom_balances = self.accounts.entry(denom)?.or_default()?;
   214	
   215	        let mut receiver_balance = denom_balances.entry(*account)?.or_default()?;
   216	        *receiver_balance = (*receiver_balance + amount).result()?;
   217	
   218	        Ok(())
   219	    }

   221	    fn escrow_coins_execute(
   222	        &mut self,
   223	        from_account: &Self::AccountId,
   224	        port_id: &PortId,
   225	        channel_id: &ChannelId,
   226	        coin: &PrefixedCoin,
   227	        _memo: &ibc::apps::transfer::types::Memo,
   228	    ) -> Result<(), TokenTransferError> {
   229	        let escrow_address = self.get_escrow_account(port_id, channel_id)?;
   230	
   231	        let denom: Denom = coin.denom.clone().try_into()?;
   232	        let mut denom_balances = self.accounts.entry(denom)?.or_default()?;
   233	
   234	        let amount: Amount = coin.amount.try_into()?;
   235	
   236	        let mut from_balance = denom_balances.entry(*from_account)?.or_default()?;
   237	        *from_balance = (*from_balance - amount).result()?;
   238	
   239	        let mut escrow_balance = denom_balances.entry(escrow_address)?.or_default()?;
   240	        *escrow_balance = (*escrow_balance + amount).result()?;

   372	    fn on_recv_packet_execute(
   373	        &mut self,
   374	        packet: &Packet,
   375	        _relayer: &Signer,
   376	    ) -> (ModuleExtras, Acknowledgement) {
   377	        let (extras, ack) = on_recv_packet_execute(self, packet);
   378	
   379	        if let Ok(data) = serde_json::from_slice::<PacketData>(&packet.data) {
   380	            if is_receiver_chain_source(
   381	                packet.port_id_on_a.clone(),
   382	                packet.chan_id_on_a.clone(),
   383	                &data.token.denom,
   384	            ) {
   385	                let incoming_transfer = extras
   386	                    .events
   387	                    .iter()
   388	                    .find(|event| {
   389	                        event.kind == "fungible_token_packet"
   390	                            && event
   391	                                .attributes
   392	                                .contains(&("success".to_string(), "true".to_string()).into())
   393	                    })
   394	                    .map(|event| -> crate::Result<TransferInfo> {
   395	                        let mut denom = data.token.denom.clone();
   396	
   397	                        denom.remove_trace_prefix(&TracePrefix::new(
   398	                            packet.port_id_on_a.clone(),
   399	                            packet.chan_id_on_a.clone(),
   400	                        ));
   401	
   402	                        let get_attr = |ev: &ModuleEvent, key: &str| {
   403	                            ev.attributes
   404	                                .iter()
   405	                                .find(|attr| attr.key == key)
   406	                                .map(|attr| attr.value.clone())
   407	                                .ok_or_else(|| {
   408	                                    crate::Error::Ibc(format!(
   409	                                        "Missing transfer event attribute {}",
   410	                                        key
   411	                                    ))
   412	                                })
   413	                        };
   414	                        Ok(TransferInfo {
   415	                            denom,
   416	                            amount: get_attr(event, "amount")?.parse()?,
   417	                            memo: get_attr(event, "memo")?,
   418	                            receiver: get_attr(event, "receiver")?,
   419	                            sender: get_attr(event, "sender")?,
   420	                        })
   421	                    })
   422	                    .transpose()
   423	                    .unwrap_or_default();
   424	
   425	                if let Some(incoming_transfer) = incoming_transfer {
   426	                    self.incoming_transfer.replace(incoming_transfer);
   427	                }
   428	            }
   429	        };
   430	        (extras, ack)
   431	    }
