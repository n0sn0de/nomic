   463	    async fn signing_commit(
   464	        &mut self,
   465	        group_index: u64,
   466	        sig_index: u64,
   467	        iteration: u32,
   468	    ) -> Result<()> {
   469	        let mut rng = thread_rng();
   470	        let config = self.get_config(group_index).await?;
   471	        let mut commitments = vec![];
   472	        for i in config.share_range(self.address)? {
   473	            let key_package: KeyPackage = self.with_key_package(|key_packages| {
   474	                Ok(key_packages
   475	                    .get((group_index, i))?
   476	                    .ok_or_else(|| {
   477	                        Error::App(format!(
   478	                            "Missing key package group {} participant {}",
   479	                            group_index, i
   480	                        ))
   481	                    })?
   482	                    .inner
   483	                    .clone())
   484	            })?;
   485	
   486	            let signing_share = key_package.signing_share();
   487	            let (nonces, commitment) = commit(signing_share, &mut rng);
   488	            self.with_signing_nonces(|signing_nonces| {
   489	                signing_nonces.insert(
   490	                    (group_index, sig_index, iteration, i),
   491	                    Adapter {
   492	                        inner: nonces.clone(),
   493	                    },
   494	                )?;
   495	
   496	                Ok(())
   497	            })?;
   498	            commitments.push(Adapter { inner: commitment });
   499	        }
   500	
   501	        let commitments: LengthVec<u16, Adapter<SigningCommitments>> = commitments.try_into()?;
   502	        self.call(|app| {
   503	            build_call!(app.frost.submit_commitments(
   504	                group_index,
   505	                sig_index,
   506	                iteration,
   507	                commitments.clone()
   508	            ))
   509	        })
   510	        .await?;
   511	
   512	        log::info!(
   513	            "Submitted commitment for group {} signature {} iteration {}",
   514	            group_index,
   515	            sig_index,
   516	            iteration
   517	        );
   518	
   519	        Ok(())
   520	    }

   522	    /// Performs the signing round 2 step for the given group index and
   523	    /// participant index.
   524	    ///
   525	    /// See [`frost_secp256k1_tr::round2::sign`].
   526	    async fn signing_sign(
   527	        &mut self,
   528	        group_index: u64,
   529	        sig_index: u64,
   530	        iteration: u32,
   531	    ) -> Result<()> {
   532	        let Some(signing_package) = self
   533	            .client()
   534	            .query(|app| {
   535	                app.frost
   536	                    .signing_package(group_index, sig_index)
   537	                    .map(|p| p.map(|p| p.inner))
   538	            })
   539	            .await?
   540	        else {
   541	            return Err(Error::App("No signing package".to_string()));
   542	        };
   543	
   544	        let config = self.get_config(group_index).await?;
   545	
   546	        let mut sig_shares = vec![];
   547	        for i in config.share_range(self.address)? {
   548	            let key_package: KeyPackage = self.with_key_package(|key_packages| {
   549	                Ok(key_packages
   550	                    .get((group_index, i))?
   551	                    .ok_or_else(|| Error::App("Missing key package".to_string()))?
   552	                    .inner
   553	                    .clone())
   554	            })?;
   555	
   556	            let nonces = self.with_signing_nonces(|signing_nonces| {
   557	                Ok(signing_nonces
   558	                    .get((group_index, sig_index, iteration, i))?
   559	                    .ok_or_else(|| Error::App("Missing signing nonces".to_string()))?
   560	                    .inner
   561	                    .clone())
   562	            })?;
   563	
   564	            let sig_share = round2::sign(&signing_package, &nonces, &key_package)
   565	                .map_err(|e| Error::App(format!("Error during signing: {}", e)))?;
   566	
   567	            sig_shares.push(Adapter { inner: sig_share });
   568	        }
   569	
   570	        let sig_shares: LengthVec<u16, Adapter<SignatureShare>> = sig_shares.try_into()?;
   571	        self.call(|app| {
   572	            build_call!(app.frost.submit_sig_shares(
   573	                group_index,
   574	                sig_index,
   575	                iteration,
   576	                sig_shares.clone()
   577	            ))
   578	        })
   579	        .await?;
   580	
   581	        log::info!(
   582	            "Submitted signature share for group {} signature {} iteration {}",
   583	            group_index,
   584	            sig_index,
   585	            iteration
   586	        );
   587	
   588	        Ok(())
   589	    }
